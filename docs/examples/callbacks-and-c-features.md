# CallbackとC固有機能

> このCコードは、altcの意味とコストモデルを説明するために推定した非規範的なlowering例です。実装はこの出力形式、内部名、helper、制御構造、最適化、コメント位置、翻訳単位構成を模倣してはならず、Draft規範仕様を満たす別のC11生成を選べます。

関連仕様：[C連携](../spec/c-interop.md)、[function pointer](../spec/declarations-and-modules.md#function-pointer)

## C headerに対応するaltc宣言

```altc
extern "C" from "interop.h" {
  opaque type Context;
  var threshold: c_int;
  const SCALE: c_int;
  macro fn apply_scale(value: c_int) -> c_int;
  fn default_context() -> *const Context;
  fn context_threshold(context: *const Context) -> c_int;
}

extern "C" from <stdlib.h> {
  fn qsort(
    base: *mut unit,
    count: usize,
    width: usize,
    compare: extern "C" fn(*const unit, *const unit) -> c_int,
  );
}

extern "C" from <stdio.h> {
  fn printf(format: *const c_char, ...) -> c_int;
}

@c_name("compare_i32_callback")
export "C" fn compare_i32(left: *const unit, right: *const unit) -> c_int {
  let lhs: *const i32 = ptr_cast[i32](left);
  let rhs: *const i32 = ptr_cast[i32](right);
  if *lhs < *rhs { -1 } else if *lhs > *rhs { 1 } else { 0 }
}

@entry
fn main() -> c_int {
  let mut values: [i32; 4] = [4, 1, 3, 2];
  qsort(
    ptr_cast[unit](&values[0]),
    4,
    sizeof[i32],
    compare_i32,
  );

  threshold = 2;
  let context: *const Context = default_context();
  let context_limit: c_int = context_threshold(context);
  let scaled: c_int = apply_scale(values[0] as c_int);
  printf(c"scale=%d first=%d\n", scaled, values[0] as c_int);
  if SCALE == 3 && scaled == 3 && context_limit == 5 { 0 } else { 1 }
}
```

## 取り得るC11

```c
/* Generated by altc. Do not edit. */
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

int threshold = 0;
#define SCALE 3
#define apply_scale(value) ((value) * SCALE)

typedef struct Context {
  int limit;
} Context;

static const Context default_context_value = { 5 };

const Context *default_context(void) {
  return &default_context_value;
}

int context_threshold(const Context *context) {
  return context->limit;
}

int compare_i32_callback(const void *left, const void *right) {
  const int32_t *lhs = (const int32_t *)left;
  const int32_t *rhs = (const int32_t *)right;
  if (*lhs < *rhs) {
    return -1;
  }
  if (*lhs > *rhs) {
    return 1;
  }
  return 0;
}

int main(void) {
  int32_t values[4] = {
    INT32_C(4), INT32_C(1), INT32_C(3), INT32_C(2)
  };
  qsort(values, 4u, sizeof(int32_t), compare_i32_callback);

  threshold = 2;
  const Context *context = default_context();
  int context_limit = context_threshold(context);
  int macro_argument = (int)values[0];
  int scaled = apply_scale(macro_argument);
  (void)printf("scale=%d first=%d\n", scaled, (int)values[0]);
  return (SCALE == 3 && scaled == 3 && context_limit == 5) ? 0 : 1;
}
```

## 対応の要点

- callbackはnon-capturing C function pointerです。closure環境やtrampolineはありません。
- callbackと`qsort` callにmemory effect annotationは不要です。`compare_i32`には`indirect-read`、`qsort` callには`extern-call`、callbackを呼ぶC側の内部は不透明な境界としてsummaryへ残ります。
- `Context`のlayoutはaltcへ公開せず、non-null pointerだけを受け渡します。
- function-like macroへはtemporaryを渡し、macro bodyがargumentを複数回使ってもaltc式を一度しか評価しません。
- variadic argumentは`c_int`へ明示変換済みです。compilerが固定幅型を暗黙promoteしません。
- `threshold`のwriteはmemory access summaryの`global-write`です。`default_context`、`context_threshold`、`apply_scale`、`printf`のcallは`extern-call`として表示できます。
