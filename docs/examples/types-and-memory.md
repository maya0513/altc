# 型とmemory

> このCコードは、altcの意味とコストモデルを説明するために推定した非規範的なlowering例です。実装はこの出力形式、内部名、helper、制御構造、最適化、コメント位置、翻訳単位構成を模倣してはならず、Draft規範仕様を満たす別のC11生成を選べます。

関連仕様：[字句構造](../spec/lexical-structure.md)、[型と値](../spec/types-and-values.md)、[memory access summary](../spec/memory-effects-and-errors.md#memory-access-summary)

## altc

```altc
extern "C" from <stdio.h> {
  fn puts(message: *const c_char) -> c_int;
}

fn first_or(values: []const i32, fallback: i32) -> i32 {
  match values.get(0) {
    nonnull(pointer) => *pointer,
    null => fallback,
  }
}

@entry
fn main() -> c_int {
  let scalar: char = 'λ';
  let text: str = "東京";
  let bytes: []const u8 = b"ABC";
  let c_text: *const c_char = c"hello from C";

  let values: [i32; 3] = [10, 20, 30];
  let view: []const i32 = values[0..3];
  let unchecked: i32 = view[1];
  let safe: i32 = first_or(view, -1);
  let absent: ?*const i32 = null;
  let address: usize = ptr_addr(&values[0]);
  let restored: ?*const i32 = ptr_from_addr[i32](address);

  let narrowed: u8 = truncate[u8](300u16);
  let casted: Option[u8] = checked_cast[u8](300u16);
  let wrapped: i32 = 2_147_483_647i32 +% 1i32;
  let saturated: u8 = 250u8 +| 20u8;
  let checked: Option[i32] = checked_add(
    2_000_000_000i32,
    1_000_000_000i32,
  );
  let promoted: c_int = safe as c_int;

  puts(c_text);
  promoted
}
```

## 取り得るC11

```c
/* Generated by altc. Do not edit. */
#include <stddef.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>
#include <string.h>

typedef struct {
  const uint8_t *data;
  size_t len;
} StringView;

typedef struct {
  const uint8_t *data;
  size_t len;
} ByteSlice;

typedef struct {
  const int32_t *data;
  size_t len;
} I32Slice;

typedef struct {
  bool is_some;
  int32_t value;
} OptionI32;

typedef struct {
  bool is_some;
  uint8_t value;
} OptionU8;

static const int32_t *slice_get(
    I32Slice values,
    size_t index) {
  return index < values.len ? &values.data[index] : NULL;
}

static int32_t first_or(
    I32Slice values,
    int32_t fallback) {
  const int32_t *pointer = slice_get(values, 0u);
  return pointer != NULL ? *pointer : fallback;
}

static int32_t wrapping_add_i32(int32_t left, int32_t right) {
  uint32_t bits = (uint32_t)left + (uint32_t)right;
  int32_t result;
  memcpy(&result, &bits, sizeof result);
  return result;
}

static uint8_t saturating_add_u8(uint8_t left, uint8_t right) {
  uint16_t sum = (uint16_t)left + (uint16_t)right;
  return sum > UINT8_MAX ? UINT8_MAX : (uint8_t)sum;
}

static OptionU8 checked_cast_u16_u8(uint16_t value) {
  if (value > UINT8_MAX) {
    OptionU8 none = { false, UINT8_C(0) };
    return none;
  }
  OptionU8 some = { true, (uint8_t)value };
  return some;
}

static OptionI32 checked_add_i32(
    int32_t left,
    int32_t right) {
  int64_t sum = (int64_t)left + (int64_t)right;
  if (sum < INT32_MIN || sum > INT32_MAX) {
    OptionI32 none = { false, INT32_C(0) };
    return none;
  }
  OptionI32 some = { true, (int32_t)sum };
  return some;
}

int main(void) {
  uint32_t scalar = UINT32_C(0x03bb);
  static const uint8_t text_data[] = {
    0xe6, 0x9d, 0xb1, 0xe4, 0xba, 0xac
  };
  StringView text = { text_data, sizeof text_data };
  static const uint8_t byte_data[] = { 'A', 'B', 'C' };
  ByteSlice bytes = { byte_data, sizeof byte_data };
  static const char c_text[] = "hello from C";

  int32_t values[3] = { INT32_C(10), INT32_C(20), INT32_C(30) };
  I32Slice view = { values, 3u };
  int32_t unchecked = view.data[1];
  int32_t safe = first_or(view, -INT32_C(1));
  const int32_t *absent = NULL;
  uintptr_t address = (uintptr_t)&values[0];
  const int32_t *restored = (const int32_t *)address;
  uint8_t narrowed = (uint8_t)UINT16_C(300);
  OptionU8 casted = checked_cast_u16_u8(UINT16_C(300));
  int32_t wrapped = wrapping_add_i32(INT32_MAX, INT32_C(1));
  uint8_t saturated = saturating_add_u8(UINT8_C(250), UINT8_C(20));
  OptionI32 checked = checked_add_i32(
      INT32_C(2000000000),
      INT32_C(1000000000));
  int promoted = (int)safe;

  (void)puts(c_text);
  (void)scalar;
  (void)text;
  (void)bytes;
  (void)unchecked;
  (void)absent;
  (void)restored;
  (void)narrowed;
  (void)casted;
  (void)wrapped;
  (void)saturated;
  (void)checked;
  return promoted;
}
```

## 対応の要点

- `str`、byte slice、通常sliceはpointerとlengthだけで、内容をcopyもallocateもしません。
- Cではnon-null pointerとnullable pointerが同じ表現です。altc type checkerが`get`のnullable resultを`nonnull(pointer)`でrefineするまでdereferenceを許さないことに意味があります。
- `ptr_from_addr`もnullable pointerを返します。元addressがnonzeroでもprovenanceやlifetimeは保証されず、この例ではdereferenceしません。
- `view[1]`はbounds branchを生成しません。`get(0)`だけが明示的にlengthを検査します。
- `first_or`と`main`はtracked-effect-freeですがpureではありません。前者のpointer readは`indirect-read`、後者の`puts` callは`extern-call`としてmemory access summaryへ現れます。
- wrapping、saturating、checked helperの形は非規範的です。compiler builtinや別のbit conversionを使えます。
- `checked_cast[u8](300u16)`とoverflowする`checked_add`は`None`になります。C例ではtagだけの小さなoptional valueで表しています。
- UTF-8 storageの並べ方、literalの重複統合、内部slice type名は実装裁量です。
- この例で最終計算に使わないlocalは、型とstorage表現を並べて示すために残しています。未使用localは意味を持たず、diagnostic policyに従ってwarningにできます。
